# Site build + local review runs.

# The referee panel itself is pinned per-run by .github/workflows/review.yml
# (it resolves the current HEAD SHA so every review records exactly which
# version produced it). For local use, install it unpinned:
#
#     pip install "peerreviewagents @ git+https://github.com/pgarrett-scripps/PeerReviewAgents.git"

# mkdocs is capped: 2.0 removes the plugin and theming systems that
# mkdocs-material depends on, with no migration path.
mkdocs>=1.6,<2

# Open Graph card per page. It also needs system Cairo — see the apt step in
# .github/workflows/publish.yml. Without these mkdocs still builds, but every
# shared review link renders as a bare URL instead of a card.
mkdocs-material[imaging]>=9.5
pyyaml>=6.0
